cmake_minimum_required(VERSION 3.16)
project(pp2_sim CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# C ABI library (pp2_load_table / pp2_detect / pp2_shoot / ...) + CLI
add_library(pp2_sim SHARED api/pp2_api.cpp)
target_compile_options(pp2_sim PRIVATE -O2 -Wno-misleading-indentation)
add_executable(pp2_cli src/pp2_cli.cpp)
target_compile_options(pp2_cli PRIVATE -O2 -Wno-misleading-indentation)
set_target_properties(pp2_cli PROPERTIES OUTPUT_NAME pp2_sim)
# P2Z live predictor host (il2cpp binding).  Static lib: compile check on arm64 (NOT run on a device yet); link it into the injected module later.
add_library(pp2_il2cpp_host STATIC android/pp2_il2cpp_host.cpp)
target_compile_options(pp2_il2cpp_host PRIVATE -O2 -Wno-misleading-indentation)
if(NOT ANDROID)
  add_executable(test_engine tests/test_engine.cpp)
  enable_testing()
  add_test(NAME engine COMMAND test_engine ${CMAKE_SOURCE_DIR}/data)
endif()
